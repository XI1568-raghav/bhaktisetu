Write failing tests in `tests/seo/generate-programmatic-pages.test.ts` BEFORE implementation:

```typescript
import { describe, it, expect } from 'vitest';
import { generatePage, slugify, validateTemplate } from '../../scripts/seo/generate-programmatic-pages';
import matrix from '../../data/seo/puja-city-language-matrix.json';

describe('programmatic SEO generator', () => {
  it('emits valid frontmatter for every (puja, city, language) combo', () => {
    for (const p of matrix.pujas) {
      for (const c of matrix.cities) {
        for (const l of matrix.languages) {
          const page = generatePage(p, c, l);
          expect(page.slug).toMatch(/^[a-z0-9-]+$/);
          expect(page.title).toHaveLength.be.greaterThan(10);
          expect(page.metaDescription).toHaveLength.lessThanOrEqual(155);
          expect(page.hindiTitle).toBeTruthy();
          expect(page.englishTitle).toBeTruthy();
        }
      }
    }
  });

  it('has zero slug collisions across full matrix', () => {
    const slugs = new Set<string>();
    for (const p of matrix.pujas) {
      for (const c of matrix.cities) {
        for (const l of matrix.languages) {
          const s = slugify(p, c, l);
          expect(slugs.has(s)).toBe(false);
          slugs.add(s);
        }
      }
    }
  });

  it('all 50 templates have meta description ≤ 155 chars', () => {
    for (const tpl of matrix.templates) {
      expect(tpl.metaDescriptionPattern.length).toBeLessThanOrEqual(155);
    }
  });

  it('renders Remix route for sample slug', () => {
    const page = generatePage('ganesh-puja', 'mumbai', 'hindi');
    expect(page.canonical).toBe(`/programmatic/ganesh-puja/mumbai/hindi`);
  });
});
```
