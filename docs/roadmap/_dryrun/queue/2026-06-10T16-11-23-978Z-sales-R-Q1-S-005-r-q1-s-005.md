describe('rankBengaluruTemples', () => {
  const fixtures = [
    { name: 'A', city: 'Bengaluru', rating: 4.5, reviewCount: 1200 },
    { name: 'B', city: 'Bengaluru', rating: 4.2, reviewCount: 800 }, // excluded — rating
    { name: 'C', city: 'Bengaluru', rating: 4.4, reviewCount: 400 }, // excluded — reviews
    { name: 'D', city: 'Mysuru', rating: 4.8, reviewCount: 2000 }, // excluded — city
    { name: 'E', city: 'Bengaluru', rating: 4.9, reviewCount: 3000 },
  ];

  it('filters rating>=4.3 and reviews>=500 and city=Bengaluru', () => {
    const out = rankBengaluruTemples(fixtures);
    expect(out.map(t => t.name)).toEqual(['E', 'A']);
  });

  it('returns top 25', () => {
    const many = Array.from({ length: 100 }, (_, i) => ({
      name: `T${i}`, city: 'Bengaluru', rating: 4.3 + (i % 7) * 0.1, reviewCount: 500 + i * 10,
    }));
    expect(rankBengaluruTemples(many).length).toBeLessThanOrEqual(25);
  });
});
