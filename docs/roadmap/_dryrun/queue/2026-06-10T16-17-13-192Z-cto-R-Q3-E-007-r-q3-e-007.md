model B2BApiKey {
  id           String    @id @default(cuid())
  orgName      String
  contactEmail String
  keyPrefix    String    @unique  // public part shown in dashboard, e.g. dk_live_abc123
  keyHash      String              // bcrypt of full key
  scopes       String[]            // ['read:temples','read:bookings',...]
  rpmLimit     Int       @default(60)
  active       Boolean   @default(true)
  lastUsedAt   DateTime?
  createdAt    DateTime  @default(now())
  revokedAt    DateTime?

  @@index([keyPrefix])
  @@index([active])
}
