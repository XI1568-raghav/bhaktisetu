### 2b. Create or edit rate-limit plugin file (path from Step 0 discovery, likely `src/plugins/rate-limit.ts`):
```typescript
import fp from 'fastify-plugin';
import rateLimit from '@fastify/rate-limit';
import { FastifyInstance, FastifyPluginOptions } from 'fastify';

export interface RateLimitOptions extends FastifyPluginOptions {
  unauthMax?: number;   // default 100
  authedMax?: number;   // default 200
  timeWindow?: string;  // default '1 minute'
}

export default fp(async (app: FastifyInstance, opts: RateLimitOptions = {}) => {
  const unauthMax = opts.unauthMax ?? 100;
  const authedMax = opts.authedMax ?? 200;
  const timeWindow = opts.timeWindow ?? '1 minute';

  await app.register(rateLimit, {
    global: true,
    max: (req) => {
      // Authenticated user gets higher allowance
      return (req as any).user?.id ? authedMax : unauthMax;
    },
    timeWindow,
    keyGenerator: (req) => {
      const userId = (req as any).user?.id;
      return userId ? `user:${userId}` : `ip:${req.ip}`;
    },
    errorResponseBuilder: (_req, context) => ({
      statusCode: 429,
      error: 'Too Many Requests',
      message: `Rate limit exceeded. Retry in ${Math.ceil(context.ttl / 1000)}s.`,
    }),
    addHeaders: {
      'x-ratelimit-limit': true,
      'x-ratelimit-remaining': true,
      'x-ratelimit-reset': true,
      'retry-after': true,
    },
    // Exempt health checks
    exclude: [
      { path: '/health' },
      { path: '/healthz' },
      { path: '/ping' },
    ],
  });
});
```
