`apps/web/e2e/refer.spec.ts`:
```ts
import { test, expect } from '@playwright/test';
test('copy link writes to clipboard', async ({ page, context }) => {
  await context.grantPermissions(['clipboard-read','clipboard-write']);
  await page.goto('/refer');
  await page.getByRole('button', { name: /copy/i }).click();
  const clip = await page.evaluate(() => navigator.clipboard.readText());
  expect(clip).toMatch(/ref=/);
});
test('whatsapp button has wa.me href', async ({ page }) => {
  await page.goto('/refer');
  const wa = page.getByRole('link', { name: /whatsapp/i });
  await expect(wa).toHaveAttribute('href', /wa\.me/);
});
```
