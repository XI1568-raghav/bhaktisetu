### 3. Weekly snapshot script — `scripts/mrr-weekly-snapshot.mjs`

new file. runs locally or in CI cron:

```javascript
#!/usr/bin/env node
// usage: node scripts/mrr-weekly-snapshot.mjs
// writes docs/finance/mrr-weekly-YYYY-MM-DD.md

import { writeFileSync, mkdirSync } from 'node:fs';
import { join } from 'node:path';

export function renderSnapshotMarkdown({ weekStart, weekEnd, total_inr, lanes, milestone_5l_reached }) {
  const fmt = (n) => `₹${n.toLocaleString('en-IN')}`;
  const rows = lanes.map(l =>
    `| ${l.lane} | ${fmt(l.mrr_inr)} | ${l.count} | ${total_inr ? ((l.mrr_inr/total_inr)*100).toFixed(1) : '0.0'}% |`
  ).join('\n');
  return `# MRR Weekly Snapshot — ${weekStart} to ${weekEnd}

**Total MTD**: ${fmt(total_inr)} / ₹5,00,000 target ${milestone_5l_reached ? '🎯 HIT' : `(${((total_inr/500000)*100).toFixed(1)}%)`}

| Lane | MRR (INR) | Count | Share |
|------|-----------|-------|-------|
${rows}

_Generated by scripts/mrr-weekly-snapshot.mjs at ${new Date().toISOString()}_
`;
}

async function main() {
  const apiUrl = process.env.API_BASE_URL || 'http://localhost:3001';
  const token = process.env.ADMIN_API_TOKEN;
  if (!token) throw new Error('ADMIN_API_TOKEN required');
  const res = await fetch(`${apiUrl}/api/v1/admin/metrics/mrr-by-lane`, {
    headers: { Authorization: `Bearer ${token}` },
  });
  const data = await res.json();
  const md = renderSnapshotMarkdown(data);
  const date = data.weekStart;
  const dir = 'docs/finance';
  mkdirSync(dir, { recursive: true });
  const path = join(dir, `mrr-weekly-${date}.md`);
  writeFileSync(path, md);
  console.log(`✅ ${path}`);

  // milestone alert
  if (data.milestone_5l_reached && process.env.RESEND_API_KEY) {
    // send email — best-effort, log if fails
    console.log('🎯 ₹5L milestone hit — sending founder alert');
    // TODO: import resend client + send
  }
}

if (import.meta.url === `file://${process.argv[1]}`) main();
```
