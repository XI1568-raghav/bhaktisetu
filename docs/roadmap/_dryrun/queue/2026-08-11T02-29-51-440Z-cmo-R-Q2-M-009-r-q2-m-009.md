## TDD (RED FIRST — superpowers doctrine)
Write failing tests in `tests/youtube/video-metadata.test.ts` BEFORE implementation:

```typescript
import { describe, it, expect } from 'vitest';
import drafts from '../../data/youtube/video-drafts.json';

describe('YouTube video drafts', () => {
  it('ships exactly 12 videos (12-week plan)', () => {
    expect(drafts.videos).toHaveLength(12);
  });

  it('each title ≤ 100 chars', () => {
    for (const v of drafts.videos) {
      expect(v.titleEnglish.length).toBeLessThanOrEqual(100);
      expect(v.titleHindi.length).toBeLessThanOrEqual(100);
    }
  });

  it('each description has hook, body, CTA, hashtags', () => {
    for (const v of drafts.videos) {
      expect(v.description).toMatch(/hook:/i);
      expect(v.description).toMatch(/CTA:/i);
      expect(v.description).toMatch(/#\w+/);
    }
  });

  it('each video has 5–30 tags', () => {
    for (const v of drafts.videos) {
      expect(v.tags.length).toBeGreaterThanOrEqual(5);
      expect(v.tags.length).toBeLessThanOrEqual(30);
    }
  });

  it('each video has Hindi + English title variant', () => {
    for (const v of drafts.videos) {
      expect(v.titleHindi).toBeTruthy();
      expect(v.titleEnglish).toBeTruthy();
      expect(v.titleHindi).not.toBe(v.titleEnglish);
    }
  });

  it('all 12 videos map to unique upload dates in editorial calendar', () => {
    const dates = drafts.videos.map(v => v.scheduledUploadDate);
    expect(new Set(dates).size).toBe(12);
  });

  it('each video has thumbnail brief + script outline + duration target', () => {
    for (const v of drafts.videos) {
      expect(v.thumbnailBrief).toBeTruthy();
      expect(v.scriptOutline.hook).toBeTruthy();
      expect(v.scriptOutline.body).toBeTruthy();
      expect(v.scriptOutline.cta).toBeTruthy();
      expect(v.durationTargetMinutes).toBeGreaterThan(0);
    }
  });
});
```
