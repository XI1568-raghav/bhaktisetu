## Implementation

### apps/api/src/plugins/cron-auth.ts
```ts
import fp from 'fastify-plugin';
import type { FastifyPluginAsync } from 'fastify';

const cronAuth: FastifyPluginAsync = async (app) => {
  app.addHook('preHandler', async (req, reply) => {
    if (!req.url.startsWith('/api/v1/cron/')) return;
    const expected = process.env.CRON_AUTH_TOKEN;
    if (!expected) {
      req.log.error('CRON_AUTH_TOKEN not set in env');
      return reply.code(500).send({ error: 'cron auth not configured' });
    }
    const got = req.headers['x-cron-auth'];
    if (got !== expected) {
      return reply.code(401).send({ error: 'unauthorized' });
    }
  });
};

export default fp(cronAuth, { name: 'cron-auth' });
```
