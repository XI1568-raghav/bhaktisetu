// edge-runtime migration completes (R-Q1-E-001 Path A) or after Vercel migration (Path B).
//
// DO NOT copy this into apps/web/ today — its presence triggers the CF Pages build
// requirement that every non-static route declare `export const runtime = 'edge'`,
// which currently fails on ~292 of 370 routes.
//
// Origin: 7adbd87 (reverted in 9c71e60). See docs/engineering/R-Q1-E-001-csp-nonce-middleware-handoff.md.

import { NextRequest, NextResponse } from 'next/server'

export function middleware(request: NextRequest) {
  // Generate a fresh nonce per request (16 random bytes, base64-encoded).
  const nonce = Buffer.from(crypto.getRandomValues(new Uint8Array(16))).toString('base64')

  // CSP with nonce + strict-dynamic. Whitelisted hosts cover Razorpay, Stripe,
  // PostHog, GTM, CF Insights, AdSense, and YouTube live darshan embed.
  const csp = [
    `default-src 'self'`,
    `script-src 'self' 'nonce-${nonce}' 'strict-dynamic' https: 'unsafe-inline'`, // 'unsafe-inline' ignored by browsers that honor nonce
    `style-src 'self' 'unsafe-inline' https://fonts.googleapis.com`,
    `img-src 'self' data: blob: https:`,
    `font-src 'self' https://fonts.gstatic.com data:`,
    `connect-src 'self' https://api.darshya.online https://*.supabase.co https://app.posthog.com https://*.cloudflareinsights.com https://api.razorpay.com https://api.stripe.com wss:`,
    `frame-src 'self' https://www.youtube.com https://youtube.com https://api.razorpay.com https://js.stripe.com https://hooks.stripe.com https://meet.jit.si`,
    `frame-ancestors 'none'`,
    `base-uri 'self'`,
    `form-action 'self' https://api.razorpay.com https://hooks.stripe.com`,
    `object-src 'none'`,
    `upgrade-insecure-requests`,
  ].join('; ')

  // Pass nonce through to RSC + page components via request header.
  const requestHeaders = new Headers(request.headers)
  requestHeaders.set('x-nonce', nonce)
  requestHeaders.set('content-security-policy', csp)

  const response = NextResponse.next({ request: { headers: requestHeaders } })
  response.headers.set('content-security-policy', csp)
  response.headers.set('x-nonce', nonce)
  return response
}

// Apply to all routes except static assets + Next internals.
export const config = {
  matcher: [
    '/((?!_next/static|_next/image|favicon.ico|robots.txt|sitemap.xml|_headers|og/.*|.*\\.(?:png|jpg|jpeg|gif|webp|svg|ico|woff|woff2)).*)',
  ],
}
